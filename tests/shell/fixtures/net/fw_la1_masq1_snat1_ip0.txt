=== iptables filter ===
-P INPUT ACCEPT
-A INPUT -j BIFROST_INPUT
-P FORWARD ACCEPT
-A FORWARD -j BIFROST_FORWARD
-P OUTPUT ACCEPT
-N BIFROST_FORWARD
-A BIFROST_FORWARD -i tailscale0 -j MARK --set-xmark 0x40000/0xff0000
-A BIFROST_FORWARD -i tailscale0 -p tcp --tcp-flags SYN,RST SYN -j TCPMSS --clamp-mss-to-pmtu
-A BIFROST_FORWARD -o tailscale0 -p tcp --tcp-flags SYN,RST SYN -j TCPMSS --clamp-mss-to-pmtu
-A BIFROST_FORWARD -i tailscale0 -j ACCEPT
-A BIFROST_FORWARD -o tailscale0 -j ACCEPT
-N BIFROST_INPUT
-A BIFROST_INPUT -p udp --dport 41641 -j ACCEPT
-A BIFROST_INPUT -i tailscale0 -j ACCEPT
=== iptables nat ===
-P PREROUTING ACCEPT
-A PREROUTING -j BIFROST_PREROUTING
-P INPUT ACCEPT
-P OUTPUT ACCEPT
-P POSTROUTING ACCEPT
-A POSTROUTING -j BIFROST_POSTROUTING
-N BIFROST_POSTROUTING
-A BIFROST_POSTROUTING -m mark --mark 0x40000/0xff0000 ! -o tailscale0 -j MASQUERADE
-A BIFROST_POSTROUTING -o tailscale0 -j MASQUERADE
-N BIFROST_PREROUTING
